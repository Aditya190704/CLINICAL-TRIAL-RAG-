@app.post("/api/match", response_model=MatchResponse)
def match(req: MatchRequest):
    try:
        return pipeline.run(req.patient_case)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
    except PermissionError as e:
        logger.error(f"Permission error during trial matching: {e}")
        raise HTTPException(
            status_code=503,
            detail="Google Cloud access denied. Check BigQuery IAM permissions.",
        )
    except RuntimeError as e:
        logger.error(f"Runtime error during trial matching: {e}")
        raise HTTPException(
            status_code=500,
            detail=f"Retrieval error: {str(e)}",
        )
    except Exception as e:
        logger.error(f"Unexpected error during trial matching: {e}", exc_info=True)
        raise HTTPException(
            status_code=500,
            detail="Internal server error occurred while matching trials.",
        )

DISCLAIMER = (
    "Research decision-support only. Not medical advice. All eligibility "
    "reasoning is limited to the supplied case and registry text; confirm "
    "with the trial's contacts before referral."
)

VERDICT_ORDER = {
    "ELIGIBLE": 0,
    "POSSIBLY_ELIGIBLE": 1,
    "EXCLUDED": 2,
}


def stage_event(ctx: InvocationContext, agent_name: str, state_delta=None):
    """Build a standard ADK event, optionally updating session state."""
    return Event(
        invocation_id=ctx.invocation_id,
        author=agent_name,
        content=types.Content(
            role="model",
            parts=[types.Part(text=f"{agent_name} completed.")],
        ),
        actions=EventActions(state_delta=state_delta or {}),
    )


class PatientParserAgent(BaseAgent):
    async def _run_async_impl(
        self, ctx: InvocationContext
    ) -> AsyncGenerator[Event, None]:
        case = ctx.session.state["patient_case"]

        profile = await asyncio.to_thread(parse_patient, case)

        # Keep ADK state JSON-serializable.
        yield stage_event(
            ctx,
            self.name,
            {"patient_profile": profile.model_dump(mode="json")},
        )


class TrialRetrievalAgent(BaseAgent):
    async def _run_async_impl(
        self, ctx: InvocationContext
    ) -> AsyncGenerator[Event, None]:
        profile = PatientProfile.model_validate(
            ctx.session.state["patient_profile"]
        )
        case = ctx.session.state["patient_case"]

        candidates = await asyncio.to_thread(
            retrieve, [profile.search_query, case[:2000]]
        )

        limit = max(0, settings.candidates_to_judge)
        candidate_records = [
            {
                "nct_id": candidate.nct_id,
                "rrf": float(candidate.rrf),
                "best_sim": float(candidate.best_sim),
                "chunk_ids": list(candidate.chunk_ids),
            }
            for candidate in candidates[:limit]
        ]

        yield stage_event(
            ctx,
            self.name,
            {"trial_candidates": candidate_records},
        )


class TrialEligibilityAgent(BaseAgent):
    """Evaluate one retrieved trial using the existing eligibility judge."""

    nct_id: str

    async def _run_async_impl(
        self, ctx: InvocationContext
    ) -> AsyncGenerator[Event, None]:
        case = ctx.session.state["patient_case"]
        profile = PatientProfile.model_validate(
            ctx.session.state["patient_profile"]
        )

        trial = trials().get(self.nct_id)
        if trial is None:
            raise RuntimeError(
                f"Retrieved trial {self.nct_id} is missing from trial storage."
            )

        criteria, summary, dropped = await asyncio.to_thread(
            judge, case, profile, trial
        )

        result = {
            "criteria": [
                item.model_dump(mode="json")
                if hasattr(item, "model_dump")
                else item
                for item in criteria
            ],
            "summary": summary,
            "dropped_unverified": dropped,
        }

        yield stage_event(
            ctx,
            self.name,
            {f"trial_judgement:{self.nct_id}": result},
        )


class ParallelEligibilityAgent(BaseAgent):
    """Run trial evaluations concurrently in bounded batches."""

    async def _run_async_impl(
        self, ctx: InvocationContext
    ) -> AsyncGenerator[Event, None]:
        candidates = ctx.session.state.get("trial_candidates", [])
        batch_size = max(1, settings.adk_concurrency)

        for start in range(0, len(candidates), batch_size):
            batch = candidates[start : start + batch_size]

            workers = [
                TrialEligibilityAgent(
                    name=f"trial_judge_{start + offset}",
                    nct_id=item["nct_id"],
                )
                for offset, item in enumerate(batch)
            ]

            parallel = ParallelAgent(
                name=f"parallel_trial_batch_{start // batch_size + 1}",
                description="Evaluate independent clinical trial eligibility.",
                sub_agents=workers,
                timeout=settings.adk_timeout_seconds,
            )

            async for event in parallel.run_async(ctx):
                yield event

        yield stage_event(ctx, self.name)


class FinalRankingAgent(BaseAgent):
    async def _run_async_impl(
        self, ctx: InvocationContext
    ) -> AsyncGenerator[Event, None]:
        state = ctx.session.state
        profile = PatientProfile.model_validate(state["patient_profile"])
        candidate_records = state.get("trial_candidates", [])
        all_trials = trials()
        all_chunks = chunks()

        evaluated = []

        for candidate in candidate_records:
            nct_id = candidate["nct_id"]
            raw = state.get(f"trial_judgement:{nct_id}")

            # Never infer eligibility if evaluation did not produce a result.
            if raw is None:
                continue

            criteria = [
                CriterionResult.model_validate(item)
                for item in raw["criteria"]
            ]
            verdict = compute_verdict(criteria)

            evaluated.append(
                (candidate, all_trials[nct_id], criteria, raw, verdict)
            )

        evaluated.sort(
            key=lambda item: (
                VERDICT_ORDER.get(item[4], 1),
                -item[0]["rrf"],
            )
        )

        matches = []

        for rank, (candidate, trial, criteria, raw, verdict) in enumerate(
            evaluated[: settings.final_results], start=1
        ):
            passages = [
                Passage(
                    chunk_type=all_chunks[chunk_id]["type"],
                    text=all_chunks[chunk_id]["text"][:700],
                )
                for chunk_id in candidate["chunk_ids"][:3]
                if chunk_id in all_chunks
            ]

            matches.append(
                TrialMatch(
                    rank=rank,
                    nct_id=trial["nct_id"],
                    title=trial["title"],
                    url=f"https://clinicaltrials.gov/study/{trial['nct_id']}",
                    status=trial["status"],
                    phases=trial.get("phases", []),
                    conditions=trial.get("conditions", [])[:6],
                    sponsor=trial.get("sponsor"),
                    locations=trial.get("locations", [])[:3],
                    relevance_score=round(candidate["best_sim"], 3),
                    verdict=verdict,
                    summary=raw["summary"],
                    criteria=criteria,
                    dropped_unverified=raw["dropped_unverified"],
                    passages=passages,
                )
            )

        response = MatchResponse(
            patient_profile=profile,
            matches=matches,
            disclaimer=DISCLAIMER,
            latency_seconds=round(
                time.time() - state["workflow_started_at"], 1
            ),
        )

        yield stage_event(
            ctx,
            self.name,
            {"match_response": response.model_dump(mode="json")},
        )


def build_workflow() -> SequentialAgent:
    """Build the request-independent ADK workflow."""
    return SequentialAgent(
        name="clinical_trial_matching_workflow",
        description=(
            "Parse a patient case, retrieve clinical trials, evaluate "
            "eligibility in parallel, and rank the results."
        ),
        sub_agents=[
            PatientParserAgent(name="patient_parser"),
            TrialRetrievalAgent(name="trial_retriever"),
            ParallelEligibilityAgent(name="eligibility_orchestrator"),
            FinalRankingAgent(name="final_ranker"),
        ],
        timeout=settings.adk_timeout_seconds,
    )


async def _run_workflow_async(patient_case: str) -> MatchResponse:
    app_name = settings.adk_app_name
    user_id = "clinical-trial-api"
    session_id = str(uuid.uuid4())

    # Request-scoped session service prevents cases from sharing session state.
    session_service = InMemorySessionService()
    await session_service.create_session(
        app_name=app_name,
        user_id=user_id,
        session_id=session_id,
        state={
            "patient_case": patient_case,
            "workflow_started_at": time.time(),
        },
    )

    runner = Runner(
        app_name=app_name,
        agent=build_workflow(),
        session_service=session_service,
    )

    message = types.Content(
        role="user",
        parts=[types.Part(text="Run the clinical trial matching workflow.")],
    )

    async for _event in runner.run_async(
        user_id=user_id,
        session_id=session_id,
        new_message=message,
    ):
        pass

    session = await session_service.get_session(
        app_name=app_name,
        user_id=user_id,
        session_id=session_id,
    )

    if session is None:
        raise RuntimeError("ADK workflow session was not found.")

    result = session.state.get("match_response")
    if result is None:
        raise RuntimeError("ADK workflow completed without a match response.")

    return MatchResponse.model_validate(result)


def run_adk_workflow(patient_case: str) -> MatchResponse:
    """Synchronous entry point for the existing synchronous FastAPI route."""
    return asyncio.run(_run_workflow_async(patient_case))

def get_state_value(state: dict[str, Any] | Any, key: str, default: Any = None) -> Any:
    """Safely get a value from state whether it is a dict or an ADK State object."""
    if hasattr(state, "get"):
        return state.get(key, default)
    if hasattr(state, "to_dict"):
        return state.to_dict().get(key, default)
    return default

# PHI regex for privacy guardrail: SSN, email, phone numbers
PHI_PATTERN = re.compile(
    r"(\b\d{3}-\d{2}-\d{4}\b|[\w.+-]+@[\w-]+\.[\w.]+|\b\d{3}[-.\s]\d{3}[-.\s]\d{4}\b)"
)


def privacy_guardrail_before_agent(ctx: Context) -> Optional[Event]:
    """Guardrail callback: Validates input privacy before agent execution.

    Rejects any case text containing identifiable personal health information (PHI)
    such as Social Security Numbers, email addresses, or phone numbers.
    """
    case = get_state_value(ctx.state, STATE_PATIENT_CASE, "")
    if not case and ctx.user_content:
        # Extract from user_content parts if state is not yet populated
        case = "".join(p.text for p in getattr(ctx.user_content, "parts", []) if hasattr(p, "text"))

    if case and PHI_PATTERN.search(case):
        logger.warning(
            f"Privacy guardrail triggered: Identifiable PHI detected in session {ctx.session.id}. Aborting."
        )
        raise ValueError(
            "Possible identifiers (SSN / email / phone) detected. Please anonymise the case."
        )
    return None


def observability_before_agent(ctx: Context) -> Optional[Event]:
    """Logs operational metadata before agent execution.

    Never logs patient narrative, credentials, or sensitive health data.
    """
    agent_name = ctx.agent_name or "unknown_agent"
    session_id = ctx.session.id if hasattr(ctx, "session") and ctx.session else "unknown_session"
    logger.info(
        f"[ADK Invocation] Agent '{agent_name}' starting | Session: {session_id} | Invocation: {ctx.invocation_id}"
    )
    # Store start timestamp in custom metadata or telemetry context
    if not hasattr(ctx, "_start_times"):
        ctx._start_times = {}
    ctx._start_times[agent_name] = time.time()
    return None


def observability_after_agent(ctx: Context) -> Optional[Event]:
    """Logs operational duration and counts after agent execution."""
    agent_name = ctx.agent_name or "unknown_agent"
    start_time = getattr(ctx, "_start_times", {}).get(agent_name)
    duration_str = f"{(time.time() - start_time):.2f}s" if start_time else "N/A"

    state_dict = ctx.state.to_dict() if hasattr(ctx.state, "to_dict") else {}
    cands = state_dict.get(STATE_CANDIDATES, [])
    evals = state_dict.get(STATE_VERIFIED_EVALUATIONS, [])

    logger.info(
        f"[ADK Invocation] Agent '{agent_name}' completed | Duration: {duration_str} | "
        f"Candidates: {len(cands) if isinstance(cands, list) else 0} | "
        f"Verified Matches: {len(evals) if isinstance(evals, list) else 0}"
    )
    return None


def age_validation_after_parser(ctx: Context) -> Optional[Event]:
    """Validates extracted patient age against original narrative.

    An extracted age must literally appear in the source text to prevent
    hallucinatory age inference by language models.
    """
    case = get_state_value(ctx.state, STATE_PATIENT_CASE, "")
    profile_dict = get_state_value(ctx.state, STATE_PATIENT_PROFILE)

    if profile_dict and isinstance(profile_dict, dict):
        age_years = profile_dict.get("age_years")
        if age_years is not None:
            try:
                age_int = int(age_years)
                if not re.search(rf"\b{age_int}\b", case):
                    logger.info(
                        f"Parser guardrail: Age {age_years} not found verbatim in narrative. Setting to None."
                    )
                    profile_dict["age_years"] = None
                    ctx.state.update({STATE_PATIENT_PROFILE: profile_dict})
            except Exception as e:
                logger.debug(f"Age verification check exception: {e}")
    return None

# Reusable BigQuery client instance
_client: Optional[bigquery.Client] = None


def get_bigquery_client() -> bigquery.Client:
    """Lazily initialize and return the BigQuery client."""
    global _client
    if _client is None:
        if not settings.gcp_project:
            raise ValueError(
                "GCP project ID is not configured. Set GCP_PROJECT in backend/.env"
            )
        _client = bigquery.Client(
            project=settings.gcp_project,
            location=settings.bq_location,
        )
    return _client


def set_bigquery_client(client: Optional[bigquery.Client]) -> None:
    """Override the BigQuery client instance (useful for unit testing)."""
    global _client
    _client = client


@dataclass
class Candidate:
    """Trial candidate with fused relevance and evidence chunk IDs."""
    nct_id: str
    rrf: float = 0.0
    best_sim: float = 0.0
    chunk_ids: list[str] = field(default_factory=list)


def validate_embeddings(vectors: list[list[float]], expected_dim: int) -> None:
    """Validate vector dimensions, non-emptiness, and numerical values."""
    if not vectors:
        raise ValueError("Cannot perform vector search with empty vectors list.")

    for idx, vec in enumerate(vectors):
        if not isinstance(vec, (list, tuple)):
            raise ValueError(f"Vector at index {idx} must be a list/sequence of numbers.")
        if len(vec) != expected_dim:
            raise ValueError(
                f"Embedding dimension mismatch at index {idx}: expected {expected_dim}, "
                f"got {len(vec)}."
            )
        for dim_idx, val in enumerate(vec):
            if val is None or math.isnan(val) or math.isinf(val):
                raise ValueError(
                    f"Vector at index {idx} contains invalid numerical value {val} at dimension {dim_idx}."
                )


def build_vector_search_sql(
    project_id: str,
    dataset_id: str,
    table_id: str,
    num_queries: int,
) -> str:
    """Construct a parameterized BigQuery VECTOR_SEARCH GoogleSQL query."""
    if num_queries <= 0:
        raise ValueError("num_queries must be greater than 0")

    # Construct the query table subquery with UNION ALL for batch queries
    subquery_parts = [
        f"SELECT {i} AS query_id, @q{i} AS query_vector"
        for i in range(num_queries)
    ]
    query_subquery = "\n      UNION ALL\n      ".join(subquery_parts)

    full_table_path = f"`{project_id}.{dataset_id}.{table_id}`"

    sql = f"""
SELECT
  query.query_id AS query_id,
  base.chunk_id AS chunk_id,
  base.nct_id AS nct_id,
  base.chunk_text AS chunk_text,
  base.chunk_type AS chunk_type,
  base.metadata_json AS metadata_json,
  distance
FROM VECTOR_SEARCH(
  TABLE {full_table_path},
  'embedding',
  (
    {query_subquery}
  ),
  query_column_to_search => 'query_vector',
  top_k => @top_k,
  distance_type => 'COSINE'
)
ORDER BY query.query_id, distance ASC
"""
    return sql.strip()


def execute_vector_search(
    client: bigquery.Client,
    vectors: list[list[float]],
    top_k: int,
) -> list[bigquery.Row]:
    """Execute VECTOR_SEARCH against the configured BigQuery table."""
    project = settings.gcp_project
    dataset = settings.bigquery_dataset
    table = settings.bigquery_chunks_table

    sql = build_vector_search_sql(
        project_id=project,
        dataset_id=dataset,
        table_id=table,
        num_queries=len(vectors),
    )

    query_params: list[bigquery.ArrayQueryParameter | bigquery.ScalarQueryParameter] = [
        bigquery.ArrayQueryParameter(f"q{i}", "FLOAT64", [float(x) for x in vectors[i]])
        for i in range(len(vectors))
    ]
    query_params.append(bigquery.ScalarQueryParameter("top_k", "INT64", top_k))

    job_config = bigquery.QueryJobConfig(
        query_parameters=query_params,
        default_dataset=f"{project}.{dataset}",
    )

    try:
        query_job = client.query(sql, job_config=job_config, location=settings.bq_location)
        rows = list(query_job.result())
        return rows
    except gcp_exceptions.NotFound as e:
        msg = (
            f"BigQuery resource not found: `{project}.{dataset}.{table}` in location "
            f"'{settings.bq_location}'. Ensure the dataset and table are created and populated. "
            f"Original error: {e}"
        )
        logger.error(msg)
        raise RuntimeError(msg) from e
    except gcp_exceptions.Forbidden as e:
        msg = (
            f"Permission denied accessing BigQuery table `{project}.{dataset}.{table}`. "
            f"Ensure your GCP credentials or Service Account has BigQuery Data Viewer and "
            f"BigQuery Job User roles. Original error: {e}"
        )
        logger.error(msg)
        raise PermissionError(msg) from e
    except gcp_exceptions.BadRequest as e:
        msg = f"BigQuery vector search query failed (BadRequest): {e}"
        logger.error(msg)
        raise RuntimeError(msg) from e
    except gcp_exceptions.GoogleAPICallError as e:
        msg = f"BigQuery vector search API call failed: {e}"
        logger.error(msg)
        raise RuntimeError(msg) from e
    except Exception as e:
        msg = f"Unexpected error during BigQuery vector search: {e}"
        logger.error(msg)
        raise RuntimeError(msg) from e


def fuse_and_deduplicate(rows: list[bigquery.Row]) -> list[Candidate]:
    """Process BigQuery vector search rows with Reciprocal Rank Fusion and NCT deduplication.

    For cosine distance, similarity is defined as max(0.0, 1.0 - distance).
    RRF score is calculated as 1.0 / (60 + rank) for the highest-ranking chunk
    of each trial per query.
    """
    if not rows:
        return []

    # Cache chunks locally in memory so pipeline.py can access text/type if needed
    try:
        chunk_cache = chunks()
    except Exception:
        chunk_cache = {}

    # Group rows by query_id
    grouped_by_query: dict[int, list[bigquery.Row]] = defaultdict(list)
    for row in rows:
        q_id = row.get("query_id", 0) if hasattr(row, "get") else getattr(row, "query_id", 0)
        grouped_by_query[q_id].append(row)

    candidates: dict[str, Candidate] = defaultdict(lambda: Candidate(nct_id=""))

    for q_id, q_rows in grouped_by_query.items():
        seen_trials_in_query: set[str] = set()

        for rank, row in enumerate(q_rows):
            cid = row.get("chunk_id") if hasattr(row, "get") else getattr(row, "chunk_id", None)
            nct = row.get("nct_id") if hasattr(row, "get") else getattr(row, "nct_id", None)
            chunk_text = row.get("chunk_text") if hasattr(row, "get") else getattr(row, "chunk_text", "")
            chunk_type = row.get("chunk_type") if hasattr(row, "get") else getattr(row, "chunk_type", "overview")
            distance = row.get("distance", 1.0) if hasattr(row, "get") else getattr(row, "distance", 1.0)

            if not nct or not cid:
                continue

            # Ensure chunk is available in chunks() cache for downstream passage extraction
            if cid not in chunk_cache and chunk_text:
                chunk_cache[cid] = {"nct_id": nct, "type": chunk_type, "text": chunk_text}

            # Convert cosine distance to cosine similarity
            # In BigQuery, COSINE distance = 1 - cosine_similarity
            similarity = max(0.0, 1.0 - float(distance))

            cand = candidates[nct]
            cand.nct_id = nct

            # Add chunk ID to evidence list if not already present
            if cid not in cand.chunk_ids:
                cand.chunk_ids.append(cid)

            # RRF score: count each trial once per query for its top-ranked chunk
            if nct not in seen_trials_in_query:
                cand.rrf += 1.0 / (60 + rank)
                seen_trials_in_query.add(nct)

            cand.best_sim = max(cand.best_sim, similarity)

    # Sort candidates by fused RRF score descending
    return sorted(candidates.values(), key=lambda c: c.rrf, reverse=True)


def retrieve(queries: list[str]) -> list[Candidate]:
    """Retrieve trial candidates matching the given patient queries using BigQuery VECTOR_SEARCH.

    Parameters:
        queries: List of query strings (e.g. [patient search query, case text]).

    Returns:
        List of Candidate objects sorted by RRF relevance descending.
    """
    valid_queries = [q.strip() for q in queries if q and q.strip()]
    if not valid_queries:
        logger.warning("Empty or whitespace-only queries provided to retrieve(); returning empty list.")
        return []

    # Generate embeddings using the existing embedding service
    vectors = embed(valid_queries, task_type="RETRIEVAL_QUERY")

    # Validate vector format and dimension
    validate_embeddings(vectors, expected_dim=settings.embedding_dim)

    # Get BigQuery client
    client = get_bigquery_client()

    # Execute search
    top_k = settings.top_k_chunks
    rows = execute_vector_search(client, vectors, top_k=top_k)

    # Fuse, deduplicate, and sort candidates
    return fuse_and_deduplicate(rows)

def _load(name: str):
    """Prefer GCS when configured, fall back to local file."""
    if settings.gcs_bucket:
        try:
            from app.services.gcs_store import download_json
            return download_json(name)
        except Exception as e:
            print(f"GCS read failed for {name} ({e}); using local copy")
    with open(settings.processed_dir / name, encoding="utf-8") as f:
        return json.load(f)

PHI = re.compile(
    r"(\b\d{3}-\d{2}-\d{4}\b|"
    r"[\w.+-]+@[\w-]+\.[\w.-]+|"
    r"\b\d{3}[-.\s]\d{3}[-.\s]\d{4}\b)"
)

ORDER = {
    "ELIGIBLE": 0,
    "POSSIBLY_ELIGIBLE": 1,
    "EXCLUDED": 2,
}


def run(case: str) -> MatchResponse:
    if PHI.search(case):
        raise ValueError(
            "Possible identifiers (SSN / email / phone) detected. "

from typing import Literal, Optional
from pydantic import BaseModel, Field


class MatchRequest(BaseModel):
    patient_case: str = Field(min_length=40, max_length=8000)


# ---- patient profile (LLM-extracted, explicit facts only) ----
class PatientProfile(BaseModel):
    age_years: Optional[float] = None
    sex: Literal["MALE", "FEMALE", "UNKNOWN"] = "UNKNOWN"
    primary_conditions: list[str] = []
    comorbidities: list[str] = []
    current_medications: list[str] = []
    prior_treatments: list[str] = []
    key_findings: list[str] = Field(default=[], description="labs, staging, biomarkers, performance status")
    search_query: str = Field(description="Condition-focused retrieval query built only from stated facts")


# ---- judge output (LLM) ----
class JudgedCriterion(BaseModel):
    criterion_type: Literal["INCLUSION", "EXCLUSION"]
    criterion_quote: str = Field(description="Verbatim, contiguous text copied from the trial criteria")
    patient_evidence_quote: str = Field(description="Verbatim text from the patient case, or empty string")
    impact: Literal["SUPPORTS", "BLOCKS", "UNKNOWN"]
    explanation: str


class JudgeOutput(BaseModel):
    criteria: list[JudgedCriterion]
    summary: str


# ---- API response ----
class CriterionResult(BaseModel):
    source: Literal["TRIAL_TEXT", "STRUCTURED"]
    criterion_type: Literal["INCLUSION", "EXCLUSION"]
    criterion_quote: str
    patient_evidence: str
    impact: Literal["SUPPORTS", "BLOCKS", "UNKNOWN"]
    explanation: str


class Passage(BaseModel):
    chunk_type: str
    text: str


class TrialMatch(BaseModel):
    rank: int
    nct_id: str
    title: str
    url: str
    status: str
    phases: list[str]
    conditions: list[str]
    sponsor: Optional[str]
    locations: list[str]
    relevance_score: float
    verdict: Literal["ELIGIBLE", "POSSIBLY_ELIGIBLE", "EXCLUDED"]
    summary: str
    criteria: list[CriterionResult]
    dropped_unverified: int
    passages: list[Passage]


class MatchResponse(BaseModel):
    patient_profile: PatientProfile
    matches: list[TrialMatch]
    disclaimer: str
    latency_seconds: float
